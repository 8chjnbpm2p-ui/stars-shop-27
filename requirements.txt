
import os
import logging
from telegram import (
    Update, InlineKeyboardButton, InlineKeyboardMarkup,
    ReplyKeyboardMarkup, KeyboardButton, WebAppInfo
)
from telegram.ext import (
    Application, CommandHandler, CallbackQueryHandler,
    MessageHandler, ConversationHandler, ContextTypes, filters
)

logging.basicConfig(level=logging.INFO)

SHOP_URL = "https://8chjnbpm2p-ui.github.io/stars-shop-27/"
SUPPORT = os.getenv("SUPPORT_USERNAME", "")
ADMIN_ID = os.getenv("ADMIN_CHAT_ID", "")

PACKAGES = {
    "50": "50 звёзд",
    "100": "100 звёзд",
    "250": "250 звёзд",
    "500": "500 звёзд",
}

CHOOSE_PACKAGE, ENTER_USERNAME = range(2)


async def start(update: Update, context: ContextTypes.DEFAULT_TYPE):
    keyboard = [
        [KeyboardButton("⭐ Открыть магазин", web_app=WebAppInfo(url=SHOP_URL))],
        [KeyboardButton("🛒 Купить звёзды"), KeyboardButton("💰 Цены")],
        [KeyboardButton("🆘 Поддержка")]
    ]
    await update.message.reply_text(
        "Добро пожаловать в Stars Shop 24! ⭐\n\n"
        "Выбери действие в меню ниже или используй /buy.",
        reply_markup=ReplyKeyboardMarkup(keyboard, resize_keyboard=True)
    )


async def prices(update: Update, context: ContextTypes.DEFAULT_TYPE):
    await update.message.reply_text(
        "⭐ Пакеты Stars Shop 24:\n\n"
        "⭐ 50 звёзд\n⭐ 100 звёзд\n"
        "⭐ 250 звёзд\n⭐ 500 звёзд\n\n"
        "Для оформления заказа напиши /buy.\n"
        "Стоимость и оплата пока не подключены."
    )


async def support(update: Update, context: ContextTypes.DEFAULT_TYPE):
    if SUPPORT:
        await update.message.reply_text(f"По вопросам заказа: {SUPPORT}")
    else:
        await update.message.reply_text(
            "Поддержка Stars Shop 24.\n"
            "Контакт владельца ещё не настроен."
        )


async def buy(update: Update, context: ContextTypes.DEFAULT_TYPE):
    buttons = [
        [InlineKeyboardButton("⭐ 50 звёзд", callback_data="pkg_50"),
         InlineKeyboardButton("⭐ 100 звёзд", callback_data="pkg_100")],
        [InlineKeyboardButton("⭐ 250 звёзд", callback_data="pkg_250"),
         InlineKeyboardButton("⭐ 500 звёзд", callback_data="pkg_500")],
    ]
    await update.message.reply_text(
        "Выбери пакет для заказа:",
        reply_markup=InlineKeyboardMarkup(buttons)
    )
    return CHOOSE_PACKAGE


async def choose_package(update: Update, context: ContextTypes.DEFAULT_TYPE):
    query = update.callback_query
    await query.answer()
    package = query.data.replace("pkg_", "")
    context.user_data["package"] = package
    await query.edit_message_text(
        f"Вы выбрали пакет: {PACKAGES[package]}.\n\n"
        "Теперь отправь свой Telegram-ник, например @username.\n"
        "Не отправляй пароль или код входа."
    )
    return ENTER_USERNAME


async def enter_username(update: Update, context: ContextTypes.DEFAULT_TYPE):
    username = update.message.text.strip()
    if not username.startswith("@") or len(username) < 2:
        await update.message.reply_text(
            "Введи Telegram-ник в формате @username."
        )
        return ENTER_USERNAME

    package = context.user_data.get("package", "")
    order_id = update.effective_user.id
    user = update.effective_user

    await update.message.reply_text(
        f"Заказ создан! ✅\n"
        f"Номер: {order_id}\n"
        f"Пакет: {PACKAGES[package]}\n"
        f"Telegram-ник: {username}\n\n"
        "Это пока заявка, а не оплаченный заказ. "
        "Оплата и доставка ещё не подключены."
    )

    if ADMIN_ID.isdigit():
        try:
            await context.bot.send_message(
                chat_id=int(ADMIN_ID),
                text=(
                    f"Новая заявка #{order_id}\n"
                    f"Пакет: {PACKAGES[package]}\n"
                    f"Ник: {username}\n"
                    f"Заказчик: @{user.username or 'без username'} "
                    f"(ID {user.id})"
                )
            )
        except Exception:
            logging.exception("Не удалось уведомить администратора")

    return ConversationHandler.END


async def cancel(update: Update, context: ContextTypes.DEFAULT_TYPE):
    await update.message.reply_text("Оформление заказа отменено.")
    return ConversationHandler.END


def main():
    token = os.getenv("BOT_TOKEN")
    if not token:
        raise RuntimeError("Добавь BOT_TOKEN в переменные окружения.")

    app = Application.builder().token(token).build()
    app.add_handler(CommandHandler("start", start))
    app.add_handler(CommandHandler("prices", prices))
    app.add_handler(CommandHandler("support", support))
    app.add_handler(CommandHandler("buy", buy))

    conversation = ConversationHandler(
        entry_points=[
            MessageHandler(
                filters.Regex("^🛒 Купить звёзды$"), buy
            )
        ],
        states={
            CHOOSE_PACKAGE: [
                CallbackQueryHandler(choose_package, pattern="^pkg_")
            ],
            ENTER_USERNAME: [
                MessageHandler(filters.TEXT & ~filters.COMMAND, enter_username)
            ],
        },
        fallbacks=[CommandHandler("cancel", cancel)],
    )
    app.add_handler(conversation)
    app.add_handler(
        MessageHandler(filters.Regex("^💰 Цены$"), prices)
    )
    app.add_handler(
        MessageHandler(filters.Regex("^🆘 Поддержка$"), support)
    )

    app.run_polling()


if __name__ == "__main__":
    main()
